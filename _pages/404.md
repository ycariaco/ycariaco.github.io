<script>
(() => {
  let seconds = {{ page.redirect_seconds | default: 8 }};
  const el = document.getElementById('countdown');
  const timer = setInterval(() => {
    seconds -= 1;
    if (el) el.textContent = seconds;
    if (seconds <= 0) {
      clearInterval(timer);
      window.location.href = "{{ site.baseurl | prepend: site.url }}";
    }
  }, 1000);
})();
</script>
